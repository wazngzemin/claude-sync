 func (o *FlowOrchestrator) Orchestrate(ctx context.Context, flowCtx *FlowContext) (<-chan *types.ChatResponse, error) {
 	flowCtx.ResultChan = make(chan *types.ChatResponse, constant.RespBuffSize)
 	go func() {
 		defer func() {
 			close(flowCtx.ResultChan)
 			if flowCtx.Cancel != nil {
 				flowCtx.Cancel()
 			}
 		}()
 
 		// ... DirectAgent / SyntaxRAG / SeatProcessor 保持不变
 
 		// 拒识裁决仍然先于 processors，干预不能绕过这一步
 		if flowCtx.IntentAndRefusal != nil && flowCtx.IntentAndRefusal.Reject {
 			fullChatDetailReject(ctx, true)
 			SendRefusalChunk(ctx, flowCtx, true)
 			flowCtx.AnswerSource = history.AnswerSourceCloudRefusal
 			flowCtx.Cancel()
 			return
 		}
 
 		SendRefusalChunk(ctx, flowCtx, false)
 		ctx_msg.SendMsg(ctx, constant.MsgKeyRefusalPass)
 
 		for _, processor := range o.processors {
 			if processor.CanHandle(ctx, flowCtx) {
 				_ = processor.Process(ctx, flowCtx)
 				return
 			}
 		}
 	}()
 	return flowCtx.ResultChan, nil
 }

package flow

import (
    "context"
    "errors"

    "code.byted.org/stone_vehicle/aidv/biz/config"
    "code.byted.org/stone_vehicle/aidv/biz/constant"
    "code.byted.org/stone_vehicle/aidv/biz/infrastructure/ctx_msg"
    "code.byted.org/stone_vehicle/aidv/biz/infrastructure/network"
    agent_list "code.byted.org/stone_vehicle/aidv/biz/service/agent_server/executor/dpfm/agent_list"
    "code.byted.org/stone_vehicle/aidv/biz/types"
    "code.byted.org/stone_vehicle/aidv/biz/types/history"
)

type InterventionProcessor struct{}

func NewInterventionProcessor() *InterventionProcessor {
    return &InterventionProcessor{}
}

func (p *InterventionProcessor) Name() string {
    return "InterventionProcessor"
}

func (p *InterventionProcessor) CanHandle(ctx context.Context, flowCtx *FlowContext) bool {
    if flowCtx.IntentAndRefusal != nil && flowCtx.IntentAndRefusal.Reject {
        return false
    }
    return flowCtx.InterventionErr == nil && flowCtx.InterventionResult != nil
}

func (p *InterventionProcessor) Process(ctx context.Context, flowCtx *FlowContext) error {
    flowCtx.ProcessorName = p.Name()
    flowCtx.AnswerSource = history.AnswerSourceCloudAgent
    ctx_msg.SendMsgKV(ctx, constant.MsgKeyAnswerSource, flowCtx.AnswerSource.String())
    return p.dispatchInterventionResult(ctx, flowCtx)
}

func (p *InterventionProcessor) dispatchInterventionResult(ctx context.Context, flowCtx *FlowContext) error {
    // 这里按规则类型进行分发适配
    if flowCtx.InterventionResult != nil && flowCtx.InterventionResult.Answer != "" {
        // 当前仅支持简单车控干预
        return p.handleVehicleControlIntervention(ctx, flowCtx, flowCtx.InterventionResult.Answer)
    }

    err := errors.New("unsupported intervention result")
    flowCtx.ResultChan <- network.NewChatError(ctx, constant.ErrorSystem, err.Error())
    return err
}

func (p *InterventionProcessor) handleVehicleControlIntervention(ctx context.Context, flowCtx *FlowContext, answer string) error {
    // 解析 rule.Answer 并补齐简单车控标记
    functionResponse := &types.FunctionResponse{}
    if err := config.Json.UnmarshalFromString(answer, functionResponse); err != nil {
        flowCtx.ResultChan <- network.NewChatError(ctx, constant.ErrorSystem, err.Error())
        return err
    }
    
    functionResponse.SoundZone = flowCtx.ChatReq.CoordinationInfo.SoundZone
    functionResponse.IsSimpleVehicleControl = true

    pushMsg, err := config.Json.MarshalToString(functionResponse)
    if err != nil {
        flowCtx.ResultChan <- network.NewChatError(ctx, constant.ErrorSystem, err.Error())
        return err
    }

    vehicleAgent := agent_list.NewAgentVehicleControl()
    responses, _, err := vehicleAgent.SimpleVehicleControl(ctx, pushMsg)
    if err != nil {
        flowCtx.ResultChan <- network.NewChatError(ctx, constant.ErrorSystem, err.Error())
        return err
    }

    for i := range responses {
        resp := responses[i]
        flowCtx.ResultChan <- &resp
    }

    flowCtx.Cancel()
    return nil
}

package flow

import (
    "context"

    "code.byted.org/stone_vehicle/aidv/biz/service/helper"
    "code.byted.org/stone_vehicle/aidv/biz/service/intervention"
    "code.byted.org/stone_vehicle/aidv/biz/types"
)

func prepareIntervention(ctx context.Context, req *types.ChatRequest) (*intervention.RuleSimpleInfo, error) {
    if req == nil {
        return nil, nil
    }

    // 1. 检查是否禁用干预
    if ctxValue := helper.GetContextValueFromContext(ctx);
        ctxValue != nil && ctxValue.ExtensionInfo != nil && ctxValue.ExtensionInfo[helper.CtxKeyDisableIntervene] == true {
        return nil, nil
    }

    // 2. 准备阶段只负责规则查询 (MySQL + VikingDB)
    return intervention.ModuleInterveneRule(ctx, req)
}

测试点
拒识优先：同一 query 同时命中拒识与干预时，只返回拒识，并确认 Planner 已取消。
干预命中：返回 refusal=false + 车控协议，并校验 FunctionResponse.IsSimpleVehicleControl=true；不进入 Planner，不发送 MsgKeyPlannerContinue。
干预 miss：继续走现有 processors / Planner 主链路，结果与改造前保持一致。
异常降级：规则服务超时、空 rule.Answer都必须自动回落原链路。
导入链路：Excel 导入同时覆盖规则主表、泛化问 Sheet、MySQL 落库和 VikingDB 同步四个环节。
