/* 新增的用户-记忆关联表 */
CREATE TABLE `user_memory_relation` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '主键ID',
  `entity_id` varchar(128) COLLATE utf8mb4_general_ci NOT NULL COMMENT '用户唯一标识',
  `memory_id` bigint unsigned NOT NULL COMMENT '关联的记忆ID (外键关联 memory_content_record.id)',
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '关联创建时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_entity_memory` (`entity_id`, `memory_id`), /* 确保一个用户和一条记忆的关联是唯一的 */
  KEY `idx_memory_id` (`memory_id`), /* 按记忆ID查询，找到所有关联用户 */
  KEY `idx_entity_id` (`entity_id`)  /* 按用户ID查询，找到他所有的记忆 */
) ENGINE=InnoDB COMMENT='用户与记忆的多对多关联表';

召回方案
一句话描述召回需求
可以根据产品设定的隐私策略，对于不同的记忆标签，进行车上乘客的最大集合、最小集合的标量召回逻辑。

举个例子：
车上坐了4个人abcd，此时a说话了，a的记忆里，有以下3条：
x，属于集合{a,b,c,d,p}
y，属于集合{a,b,c}
z，属于集合{a,b,d}
根据策略有两种召回情况：
对于指定标签，支持最大集合abcd的召回，即所召回的记忆，只能包括包括a的abcd的子集，比如{a},{a,b},{a,b,c},{a,c}等，此时可以召回记忆y，z。
对于指定标签，支持最小集合abcd的召回，即所召回的记忆，只能包括包括abcd的全集，这种情况下，可以召回记忆x。
整体召回链路概览

核心实现
核心需要实现的是对集合的子集、超集判断，这块逻辑不管是放在向量数据库还是数据库本身，都比较难以平衡效率和结果，所以考虑放在召回链路中，从代码中层面实现。
考虑使用三方库：https://github.com/deckarep/golang-set，实现对于超集和子集的判断逻辑：

setA := set.NewSet("1", "2", "3")         
setB := set.NewSet("1", "2", "3", "4")         
setC := set.NewSet("1", "2", "5")          
fmt.Println(setA.IsSubset(setB))    // true
fmt.Println(setA.IsSubset(setC))    // false
fmt.Println(setB.IsSuperset(setA))  // true
车上最多5个faceid用户，集合中必须包括说话人的元素，所以最大情况下的枚举是15种，记忆召回的个数为N，那么整体的时间复杂度是：O（15N），按照当下N<=1000评估，总的耗时在微秒级别，对主链路基本没有影响。
