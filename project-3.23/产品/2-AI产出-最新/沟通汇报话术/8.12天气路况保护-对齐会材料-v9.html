// 每个被监控的 key 维护 {lastRaw, stableCount, confirmed}
const DEBOUNCE_PERIODS = 3;  // v4 定稿：连续3个上报周期(=15s)一致才确认变更

function onVLMReport(scene) {              // scene = vision.outside.scene
  const cur = normalize(scene["天气情况描述"]); // 只取枚举位，不看全文
  if (cur === state.lastRaw) {
    state.stableCount++;
  } else {
    state.lastRaw = cur;
    state.stableCount = 1;
  }
  if (state.stableCount >= DEBOUNCE_PERIODS && cur !== state.confirmed) {
    emit({ event: "weather_changed", from: state.confirmed, to: cur, ts: now() });
    state.confirmed = cur;              // one-shot：同值不再发；切出后解锁
  }
}

function onTimeout() {                 // 本轮无上报：不计数、不改 confirmed、不触发
}
