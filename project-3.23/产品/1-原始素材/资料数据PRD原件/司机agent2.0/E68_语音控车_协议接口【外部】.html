协议消息总览

本文档定义座舱语音模块与智驾模块之间的通信协议，采用 protobuf3 格式。座舱侧将用户语音解析为结构化意图（VLARequest），智驾侧返回执行结果（VoiceCommandResp）或 CoT 推理内容（VLAtext）。

message DrivingRequest {
    int32 id =1;// 指令ID，不唯一
    int64 timestamp =2;  // 时间戳，如1720181322923，可解析为ETC时间
    DrivingType type = 3;// 指令类别
    DrivingCommand command = 4;// 指令内容
}

enum DrivingType {// 指令类别（待补充）
  CONTROL = 0;       // 行车控制指令
  VLA_Voice = 1;           // VLA语音控车指令
  E2E = 2;           // E2E相关指令
  VLA_Memory = 3;          //熟路模式，指令内容复用VLA指令
}

message DrivingCommand {// 指令内容（待补充）
    VLARequest vla_request = 1; // VLA相关指令内容 
    DestinationE2E destination_e2e = 2; // E2E相关指令内容

// topic： /seres/driving_response, 行车-功能指令 事件型 (ADC --> CDC)
message DrivingResponse {
    int32 id = 1;// 通用指令ID做的回复
    int64 timestamp =2; // 时间戳
    string content = 3;  // 回复反馈内容，可不填
    Status status = 4;// 反馈状态
    DrivingType type = 5; // 反馈的指令类别
    DrivingFeedback feedback = 6;// 自定义反馈消息，按需定义

    enum Status{
        success=0;// 成功
        pending=1;// 等待
        warning=2;// 异常
        failed=3; // 失败
    }
}

message DrivingFeedback {
    VoiceCommandResp voice_command_resp = 1; // VLA相关指令反馈
}
