package vla.decision;

  // 若为 0，则使用系统默认步长（如 5km/h）
  uint32 step_size_kph = 1; 
}

// 目标速度设置参数
message SpeedSetParams {
  double target_speed_kph = 1;
}

// 变道参数
message LaneChangeParams {
  LaneChangeDirection direction = 1;
}


// 变道方向
enum LaneChangeDirection {
  LC_NONE = 0;
  LC_LEFT = 1;
  LC_RIGHT = 2;
  LC_CANCEL = 3; // 取消变道
}

enum TurnActionType {
  TURN_NONE = 0;
  TURN_LEFT = 1;
  TURN_RIGHT = 2;
  TURN_U_TURN = 3;
}

// 驾驶风格
enum DrivingStyleType {
  STYLE_STANDARD = 0;
  STYLE_COMFORT = 1;     // 舒适/保守
  STYLE_SPORT = 2;       // 运动/激进
}

// 避让/微调方向
enum NudgeDirection {
  NUDGE_NONE = 0;
  NUDGE_LEFT = 1;
  NUDGE_RIGHT = 2;

// [新增] 横向微调参数
message NudgeParams {
  NudgeDirection direction = 1;

// [新增] 靠边停车参数

}

syntax = "proto3";

// ==============================
// req参数消息体定义
// ==============================

// 查询目标类型
enum QueryTargetType {
  QUERY_TARGET_UNKNOWN = 0;      // 未知
  QUERY_TARGET_UNLIMITED = 1;    // 不做限制
  QUERY_TARGET_PSD_ONLY = 2;     // 只查车位
  QUERY_TARGET_POI_ONLY = 3;     // 只查POI
}

// 地图ID筛选条件
message MapIdReqInfo {
  uint32 map_id = 1;             // 地图ID
  uint32 valid = 2;              // 是否有效: 0-忽略此条件, 1-必须匹配此map_id, 2-必须不匹配此map_id
}

// 楼层筛选条件
message FloorReqInfo {
  float floor = 1;               // 楼层信息，0层代表地面平层，负数代表地下层，正数代表地上层
  uint32 valid = 2;              // 是否有效: 0-忽略此条件, 1-必须匹配此楼层, 2-必须不匹配此楼层
}

// 子区域筛选条件
message SubRegionReqInfo {
  string sub_region = 1;         // 子区域标识，如"A"
  uint32 valid = 2;              // 是否有效: 0-忽略此条件, 1-必须匹配此子区域, 2-必须不匹配此子区域
}

// POI筛选条件
message PoiReqInfo {
  uint32 poi_type = 1;           // POI类型: 0-UNKNOWN, 
  uint32 poi_type_valid = 2;     // POI类型是否有效: 0-忽略, 1-必须匹配, 2-必须不匹配
  string poi_name = 3;           // POI名称，例如"1号电梯口"等
  uint32 poi_name_valid = 4;     // POI名称是否有效: 0-忽略, 1-必须匹配, 2-必须不匹配
}

// 车位筛选条件
message PsdReqInfo {
  uint32 psd_type = 1;           // 车位属性类型: 0-UNKNOWN,  2-CHARGING, 
  uint32 psd_type_valid = 2;     // 车位类型是否有效: 0-忽略, 1-必须匹配, 2-必须不匹配
  string psd_name = 3;           // 车位名称，例如车位编号等
  uint32 psd_name_valid = 4;     // 车位名称是否有效: 0-忽略, 1-必须匹配, 2-必须不匹配
}

// 单个查询条件组（fields之间是与的关系）
message ReqInfo {
  uint32 type = 1;                   // 查询目标类型，参考QueryTargetType枚举
  MapIdReqInfo map_id = 2;           // 地图ID筛选条件
  FloorReqInfo floor = 3;            // 楼层筛选条件
  SubRegionReqInfo sub_region = 4;   // 子区域筛选条件
  PoiReqInfo poi = 5;                // POI筛选条件
  PsdReqInfo psd = 6;                // 车位筛选条件
  uint32 parking_in_cmd = 7;         // 0 不泊入车位，1 需要泊入车位
  uint32 parking_roaming_cmd = 8;    // 0 不激活漫游，1 激活漫游
}

// topic: /m/hmi/mff/vla/pnp/query/request
// 查询请求（多个查询条件组，元素间是或的关系）
message PNPQueryRequest {
  VlaHeader header = 1;
  uint64 request_id = 2;             // 唯一请求标识符
  repeated ReqInfo req_infos = 3;    // 查询条件列表，多个条件之间是或的关系
  string req_desc = 4;               // 请求描述信息，便于调试使用
}

// ==============================
// resp参数消息体定义
// ==============================

// 查询状态码
enum PNPStatusCode {
  PNP_STATUS_SUCCESS = 0;
  PNP_STATUS_NOT_FOUND = 1;
  PNP_STATUS_INVALID_MAP = 2;
  PNP_STATUS_INVALID_PARAMS = 3;
  PNP_STATUS_INTERNAL_ERROR = 4;
}

// 状态码
message PNPStatus {
  uint32 value = 1;                  // 状态码值，参考PNPStatusCode枚举
}

// POI信息
message PoiInfo {
  uint32 poi_id = 1;                 // POI唯一标识ID
  float floor = 2;                   // 楼层信息
  string sub_region = 3;             // 子区域标识，如"A"
  uint32 poi_type = 4;               // POI类型，参考PoiReqInfo中poi_type说明
  string poi_name = 5;               // POI名称
}

// 车位信息
message ParkingSpace {
  uint32 psd_id = 1;                 // 车位唯一标识ID
  float floor = 2;                   // 楼层信息
  string sub_region = 3;             // 子区域标识，如"A"
  uint32 psd_type = 4;               // 车位属性类型，参考PsdReqInfo中psd_type说明
  string psd_name = 5;               // 车位名称
}

// 单个查询条件的响应信息
message RespInfo {
  PNPStatus error_code = 1;              // 错误状态码
  string error_msg = 2;                  // 错误信息描述
  repeated PoiInfo pois = 3;             // 匹配的POI列表（按距离从近到远排序）
  repeated ParkingSpace spaces = 4;      // 匹配的车位列表（按距离从近到远排序）
  int32 total_poi_count = 5;             // 总POI数量
  int32 total_space_count = 6;           // 总车位数量
}

// topic: /m/mff/hmi/vla/pnp/query/response
// 查询响应（对应多个查询条件的响应）
message PNPQueryResponse {
  VlaHeader header = 1;
  uint64 request_id = 2;                 // 唯一请求标识符，与请求中的request_id相同
  PNPStatus status = 3;                  // 整体查询状态（当所有req_infos查询都失败时为failed）
  repeated RespInfo resp_infos = 4;      // 响应信息列表，与PNPQueryRequest中的req_infos一一对应
  string req_desc = 5;                   // 请求描述信息，便于调试使用
}
