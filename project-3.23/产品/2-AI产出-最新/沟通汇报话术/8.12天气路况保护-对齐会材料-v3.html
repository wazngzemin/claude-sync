// 状态机：每条被监控的 key 维护 {lastRaw, stableCount, confirmed}
const DEBOUNCE_PERIODS = 2;  // 连续2个上报周期(=10s)一致才算变更

function onVLMReport(scene) {          // scene = vision.outside.scene
  const cur = normalize(scene["天气情况描述"]);  // 只取枚举位，不看全文
  if (cur === state.lastRaw) {
    state.stableCount++;
  } else {
    state.lastRaw = cur;
    state.stableCount = 1;
  }
  if (state.stableCount >= DEBOUNCE_PERIODS && cur !== state.confirmed) {
    emit({ event: "weather_changed", from: state.confirmed, to: cur, ts: now() });
    state.confirmed = cur;               // one-shot：改 confirmed 后同值不再发
  }
}

  // 什么都不做：不动 confirmed、不计数、不触发（超时 ≠ 无雨，安全红线）
}
