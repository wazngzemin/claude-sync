from transformers import AutoTokenizer, AutoModel, AutoModelForCausalLM
import torch
import pandas as pd

label_list = set()
#读取token.txt
with open("token.txt", "r") as f:
    for line in f.readlines():
        label_list.add(line.strip())

print(len(label_list))
# 载入待修改的模型基座
path = "/ai-car-vepfs1/ai_car/zhangsonglong/Models/Qwen3-1.7B/"
tokenizer = AutoTokenizer.from_pretrained(path, padding_side = "left")
model = AutoModelForCausalLM.from_pretrained(path, trust_remote_code=True)
# 标签列表

# 将所有标签加入到tokenizer中
for label in label_list:
    tokenizer.add_tokens(label)
    
# 调整embedding层和lm_head的大小以适应扩展后的tokenizer，此操作保留现有token的训练权重，仅为新token添加权重
origin_emb_size = model.get_input_embeddings().weight.data.shape[0]
model.resize_token_embeddings(max(len(tokenizer),origin_emb_size))

# 获取模型的输入嵌入层和lm_head的参数
embedding_layer = model.get_input_embeddings()
lmhead_layer = model.get_output_embeddings()

# 选择参考token：使用<|im_start|> token
reference_token_id = tokenizer.convert_tokens_to_ids("<|im_start|>")

# 将参考token的嵌入权重复制到新token
for token in label_list:
    token_id = tokenizer.convert_tokens_to_ids(token)
    embedding_layer.weight.data[token_id] = embedding_layer.weight.data[reference_token_id].clone()
    # 对于tie_word_embeddings==True的方法，由于Embedding和lm_head的权重绑定，因此无需额外初始化
    if not model.config.tie_word_embeddings:
        lmhead_layer.weight.data[token_id] = lmhead_layer.weight.data[reference_token_id].clone()
    
# 存储模型
final_model_path = "/ai-car-vepfs1/ai_car/zhangsonglong/Models/Qwen3-1.7B-extendtoken-v2/"
model.save_pretrained(final_model_path)
tokenizer.save_pretrained(final_model_path)
